System
 └── DimensionDefinition
       ├── dimKey
       ├── system
       ├── dimensions[]          (identity axes)
       │      ├─ product
       │      ├─ menu
       │      ├─ channel
       │      ├─ feature
       │      └─ function (always implicit)
       ├── arrangements[]       (scope axes)
       │      ├─ account
       │      ├─ portfolio
       │      └─ region
       └── (controls how grants are interpreted)


Client
 └── User
       ├── userId
       ├── clientId
       └── mode   →  ROLE | CUSTOM

Client
 └── Role
       ├── roleId
       └── clientId


RoleDimensionGrant
 ├── clientId
 ├── roleId
 ├── system
 ├── function
 │      ├── code
 │      ├── permissionMask
 │      └── limit
 ├── dimensions { dynamic }
 │      ├── product
 │      ├── menu
 │      ├── channel
 │      └── ...
 ├── arrangements { dynamic }
 │      ├── account
 │      ├── portfolio
 │      └── ...
 └── createdAt


UserDimensionOverride   (CUSTOM users only)
 ├── clientId
 ├── userId
 ├── system
 ├── function
 │      ├── code
 │      ├── permissionMask
 │      └── limit
 ├── dimensions { dynamic }
 ├── arrangements { dynamic }
 └── createdAt


EffectiveEntitlement
 ├── clientId
 ├── userId
 ├── system
 ├── functionCode
 ├── sourceMode        → ROLE | CUSTOM
 ├── roleId?           (only when ROLE)
 ├── effectiveMask
 ├── effectiveLimit   (winner)
 ├── dimensions { dynamic }
 ├── arrangements { dynamic }
 └── generatedAt


Trace
 ├── clientId
 ├── userId
 ├── functionCode
 ├── dimensions
 ├── arrangements
 ├── entries[]
 │      ├── source     (ROLE or CUSTOM)
 │      ├── roleId?
 │      ├── mask
 │      └── limit
 └── generatedAt
